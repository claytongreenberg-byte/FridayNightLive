// ─── QUICK SCORE (accepts # or name) ─── function quickScore(type){ const sp=g(); if(sp.gameState==='FINAL'){ toast('Game is final'); return; } const side=sp.poss; const team=getTeam(side); if(!team){ toast('No team'); return; } let pts=0; if(type==='td') pts=6; else if(type==='fg') pts=3; else if(type==='pat') pts=1; else if(type==='safety') pts=2; // Prompt for player number (with clear instruction) const input = prompt(`Enter player # for ${type.toUpperCase()} (e.g., 7) or leave blank for team score:`); let playerName = team.name; let player = null; if(input && input.trim()){ // Try to find player by number first const num = input.trim(); player = team.pl.find(p => p.n === num); if(player){ playerName = player.na; } else { // If no match by number, try by name (fallback) player = team.pl.find(p => p.na.toLowerCase() === num.toLowerCase()); if(player){ playerName = player.na; } else { // If still no match, use the input as-is playerName = num; } } } if(side==='home') sp.hs+=pts; else sp.as+=pts; sp.scoringLog.push({type,team:side,player:playerName,points:pts,time:sp.gc.rem,quarter:sp.q,timestamp:Date.now()}); // Update player stats if we found a player if(player){ if(!sp.playerStats[player.id]) sp.playerStats[player.id]={}; const st=sp.playerStats[player.id]; if(type==='td') st.tds=(st.tds||0)+1; else if(type==='fg') st.fgs=(st.fgs||0)+1; else if(type==='pat') st.pats=(st.pats||0)+1; else if(type==='safety') st.safeties=(st.safeties||0)+1; toast(`${playerName} (#${player.n}) scored!`); } else if(input && input.trim()){ toast(`${playerName} scored!`); } else { toast(`${team.name} scored!`); } pushUndo(); saveState(); renderAll(); updateScorebug(); } // ─── QUICK PENALTY (accepts # or name) ─── function quickPenalty(type){ const sp=g(); if(sp.gameState==='FINAL'){ toast('Game is final'); return; } const team=getTeam(sp.poss); if(!team){ toast('No team'); return; } const input = prompt(`Enter player # for ${type.replace('_',' ')} penalty (e.g., 55) or leave blank:`); let playerName = team.name; let player = null; if(input && input.trim()){ const num = input.trim(); player = team.pl.find(p => p.n === num); if(player){ playerName = player.na; } else { player = team.pl.find(p => p.na.toLowerCase() === num.toLowerCase()); if(player){ playerName = player.na; } else { playerName = num; } } } const yd=parseInt(prompt('Yardage (0 if none):')||'0'); sp.penaltyLog.push({type,team:sp.poss,player:playerName,yardage:yd||0,time:sp.gc.rem,quarter:sp.q,timestamp:Date.now()}); if(player){ if(!sp.playerStats[player.id]) sp.playerStats[player.id]={}; sp.playerStats[player.id].penalties=(sp.playerStats[player.id].penalties||0)+1; toast(`Penalty on ${playerName} (#${player.n})`); } else if(input && input.trim()){ toast(`Penalty on ${playerName}`); } else { toast(`Penalty on ${team.name}`); } pushUndo(); saveState(); renderAll(); }
